# 🚀 快速上手

一条 LangGraph 流程就这五步：定义状态、写节点、加边、编译、跑起来。

```bash
pnpm add langchain @langchain/core @langchain/langgraph @langchain/ollama readline-sync zod
```

<img src="https://xiejie-typora.oss-cn-chengdu.aliyuncs.com/2025-11-06-064639.png" alt="快速上手流程" style="zoom:50%;" />

```js
import * as z from "zod";
import { BaseMessage, HumanMessage } from "@langchain/core/messages";
import { ChatOllama } from "@langchain/ollama";
import {
  ChatPromptTemplate,
  MessagesPlaceholder,
  SystemMessagePromptTemplate,
} from "@langchain/core/prompts";
import { StateGraph, START, END } from "@langchain/langgraph";
import readlineSync from "readline-sync";

process.stdout.setDefaultEncoding?.("utf8");

// 1. 用 zod 规定整张图共享的状态
const Schema = z.object({
  messages: z.array(z.custom()),
  llmCalls: z.number().optional(),
});

// 2. 模型
const model = new ChatOllama({
  model: "llama3",
  temperature: 0.5,
});

// 3. 提示词。messages 占位符里放历史对话
const pt = ChatPromptTemplate.fromMessages([
  SystemMessagePromptTemplate.fromTemplate(
    "你是一个健谈的中文 AI 助手，请结合上下文尽可能详细地使用中文回答用户问题。",
  ),
  new MessagesPlaceholder("messages"),
]);

const chain = pt.pipe(model);

// 4. 节点只返回要更新的字段：追加这条回复，并把调用次数加 1
async function llmNode(state) {
  const result = await chain.invoke({
    messages: state.messages,
  });
  return {
    messages: [...state.messages, result],
    llmCalls: (state.llmCalls ?? 0) + 1,
  };
}

// 5. START -> llm -> END
const app = new StateGraph(Schema)
  .addNode("llm", llmNode)
  .addEdge(START, "llm")
  .addEdge("llm", END)
  .compile();

async function main() {
  let messages = [];
  let llmCalls = 0;

  console.log("聊天机器人已启动，输入 exit 退出。\n");

  while (true) {
    const input = readlineSync.question("你：");
    if (input === "exit") break;

    messages = [...messages, new HumanMessage(input)];
    const result = await app.invoke({ messages, llmCalls });

    messages = result.messages;
    llmCalls = result.llmCalls ?? 0;

    const last = result.messages[result.messages.length - 1];
    console.log("智能体：", last.content, "\n");
  }

  console.log(`结束对话，总共调用 LLM 次数：${llmCalls}`);
}

main();
```
