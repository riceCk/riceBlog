# 🧭 Command

节点除了改状态，还可以当场决定下一步去哪。返回一个 `Command`：

- `update`：这次要写进状态的字段
- `goto`：接下来跳到哪个节点

`addNode` 的第三个参数 `ends` 是允许跳到的节点名单。写在这里，图在编译时就知道这条边可能去哪。

```js
import { Command } from "@langchain/langgraph";

graph.addNode(
  "myNode",
  (state) => {
    return new Command({
      update: { foo: "bar" },
      goto: "myOtherNode",
    });
  },
  { ends: ["B", "C"] },
);
```

## 订单分流

A 做基础校验，数码订单去 B，实物订单去 C，最后都到 D 汇总。

```js
import { StateGraph, START, END, Command } from "@langchain/langgraph";
import * as z from "zod";
import { registry } from "@langchain/langgraph/zod";

const Schema = z.object({
  orderType: z.enum(["digital", "physical"]).optional().describe("订单类型"),
  basicInfo: z.string().optional().describe("基础校验"),
  digitalInfo: z.string().optional().describe("数码产品校验"),
  physicalInfo: z.string().optional().describe("实物产品校验"),
  logs: z.array(z.string()).register(registry, {
    reducer: {
      fn: (prev, next) => prev.concat(next),
    },
    default: () => [],
  }),
});

const app = new StateGraph(Schema)
  .addNode(
    "A",
    (state) => {
      console.log("运行 A：基础校验");
      const next = state.orderType === "digital" ? "B" : "C";
      return new Command({
        update: {
          basicInfo: "A 节点基础校验已完成",
          logs: [`A 校验完成，订单类型 ${state.orderType}，前往 ${next}`],
        },
        goto: next,
      });
    },
    { ends: ["B", "C"] },
  )
  .addNode(
    "B",
    () => {
      console.log("运行 B：数码产品校验");
      return new Command({
        update: {
          digitalInfo: "B 节点已经对数码产品做了校验",
          logs: ["数码产品订单校验完毕"],
        },
        goto: "D",
      });
    },
    { ends: ["D"] },
  )
  .addNode(
    "C",
    () => {
      console.log("运行 C：实物产品校验");
      return new Command({
        update: {
          physicalInfo: "C 节点已经对实物产品做了校验",
          logs: ["实物产品订单校验完毕"],
        },
        goto: "D",
      });
    },
    { ends: ["D"] },
  )
  .addNode("D", (state) => {
    console.log("运行 D：汇总");
    const summary =
      state.orderType === "digital"
        ? `D：数码商品流程完成：${state.digitalInfo}`
        : `D：实物商品流程完成：${state.physicalInfo}`;
    return {
      logs: [summary, "处理完成"],
    };
  })
  .addEdge(START, "A")
  .addEdge("D", END)
  .compile();

const result = await app.invoke({
  logs: [],
  orderType: "digital",
});
console.log(result);
```
