实战练习 
和大模型进行对话，大模型可以记住前面的对话内容
```bash
import readlineSync from "readline-sync";
import { ChatOllama } from "@langchain/ollama";
import {
  ChatPromptTemplate,
  MessagesPlaceholder,
  SystemMessagePromptTemplate,
  HumanMessagePromptTemplate,
} from "@langchain/core/prompts";
import { InMemoryChatMessageHistory } from "@langchain/core/chat_history";
import { HumanMessage, AIMessage } from "@langchain/core/messages";

const model = new ChatOllama({
  model: "llama3",
  temperature: 0.7,
});

# 提示词
const pt = ChatPromptTemplate.fromMessages([
  SystemMessagePromptTemplate.fromTemplate(
    `你是一个健谈中文Ai助手，请结合上下文经可能详细的使用中文回答用户问题。`,
  ),
  new MessagesPlaceholder("history"), # 会话的历史记录，一开始是个占位符
  HumanMessagePromptTemplate.fromTemplate("{input}"),
]);

# 存储会话历史
const history = new InMemoryChatMessageHistory();

# 创建一个chain
const chain = pt.pipe(model);

async function chatLoop() {
  console.log("开始会话,输入内容后回车；输入/clear清空历史，/exit 退出");

  while (true) {
    const input = readlineSync.question("用户：").trim();
    if (!input) continue;

    if (input === "/exit") {
      console.log("拜拜");
      break;
    }

    if (input === "/clear") {
      await history.clear();
      console.log("历史记录");
      continue;
    }
    let fullRes = ""; # 记录完整信息
    try {
      const values = {
        input, # 用户本地输入
        history: await history.getMessages(), # 获取之前的会话记录占位符
      };
      const stream = await chain.stream(values);
      process.stdout.write("助理：");
      for await (const chunk of stream) {
        process.stdout.write(chunk.content);
        fullRes += chunk.content;
      }
      console.log("\n");
    } catch (err) {
      console.log(err);
    }
    await history.addMessages([
      new HumanMessage(input),
      new AIMessage(fullRes),
    ]);
  }
}

chatLoop();
```

# 实现长程记忆
如何实现长程记忆？
* 1.对之前的会话记录进行总结
* 2.将会话记录存储至向量数据库

## 实现步骤
1.构建摘要链：专门对用户的会话记录做摘要操作
2.使用摘要链


**搭建摘要模型**
`summaryChain.js`其实就是完善提示词模版摘要文案，让大模型输出
```bash
import { ChatOllama } from "@langchain/ollama";
import {
  ChatPromptTemplate,
  SystemMessagePromptTemplate,
  HumanMessagePromptTemplate,
} from "@langchain/core/prompts";
import { StringOutputParser } from "@langchain/core/output_parsers";
import { RunnableSequence } from "@langchain/core/runnables";

const model = new ChatOllama({
  model: "llama3",
  temperature: 0.1, # 摘要操作，不需要模型思维发散，做低温处理
});

# 2.提示词 - 关键
const prompt = ChatPromptTemplate.fromMessages([
  SystemMessagePromptTemplate.fromTemplate(`
  你是一个「滚动摘要维护器」。你的唯一任务是将新对话内容融入已有摘要，更新用户的长期档案。

  【输入格式】
  你会收到两段内容：
  1. 「现有摘要」：之前整理好的用户信息摘要（可能为空）。
  2. 「新对话内容」：本次会话中用户和AI的完整对话记录。

  【工作流程】
  1. 从「新对话内容」中提取长期有用信息：用户偏好、个人背景、长期目标、工作中的关键进展或决策。
  2. 将提取的新信息与「现有摘要」融合。
  3. 如果新信息与旧信息冲突，以最新的信息为准。
  4. 去除一次性细节（如具体数字、临时任务）和冗余寒暄，保持摘要精简。

  【输出规则】
  - 语言：只输出简体中文。
  - 形式：只输出最终摘要文本本身，不要任何说明、前后缀、Markdown、引号或额外符号。
  - 长度：150-250字，保留最关键的信息。
  - 若新对话中没有可提取的长期信息（纯寒暄或测试性消息），直接输出原有摘要，不做任何修改。

  【目标】
  生成一份简洁、准确、一致的用户档案摘要，用于长期对话记忆。
  `),
  HumanMessagePromptTemplate.fromTemplate(`
    现有摘要：
    {summary}

    新对话内容：
    {new_lines}

    请输出“更新后的摘要”
    `),
]);

# 3.轻度清洗，兜底去除潜在前后缀、空行
const stripNoise = (text) =>
  text
    # 常见多余标记清理（可按需扩展）
    .replace(/^(```\w*|```)+/g, "")
    .replace(/```$/g, "")
    .replace(/^(更新后的摘要[::])\s*/, "")
    .trim();

const parser = new StringOutputParser();

# 构建摘要链
export const summaryChain = RunnableSequence.from([
  prompt,
  model,
  parser,
  (str) => stripNoise(str),
]);
```

`index.js`引入
```bash
const chatModel = new ChatOllama({ model: "llama3", temperature: 0.4 });

const chatPrompt = ChatPromptTemplate.fromMessages([
  SystemMessagePromptTemplate.fromTemplate(`
    你是一名中文对话助手，无论用户使用何种语言，**必须始终使用【简体中文】回答**。
    在作答前，先阅读下面可能提供的“长期摘要”，保证表述与历史一致、信息不自相矛盾。
    {maybe_summary_section}
    
    严格遵守：
    - 只用简体中文回答；除代码块外，禁止出现英文句子或段落。
    - 直接给出答案与步骤，避免客套和重复提示。
    - 列出请用精炼中文要点；必要时给出简短结论->理由->建议的结构
    - 若需给代码，**代码内注释与说明也用中文**（保留必要的英文关键字/标识符即可）。
    - 问题含糊时，依据摘要与本轮上下文做最合理假设并继续回答，**不要反问**。

    示例（用于风格约束）：
    - ☑️“可以。建议你这么做：…… 然后…… 最后……”
    - ☑️“原因是：……；因此建议：……”
    - ❌：“Sure,here is ……”
    - ❌：“Of course! ……”
   `),
  HumanMessagePromptTemplate.fromTemplate("{input}"),
]);

let summary = "";

const histroy = new InMemoryChatMessageHistory();

const chatChain = RunnableSequence.from([
  new RunnableLambda({
    func: async ({ input }) => {
      await histroy.addMessage(new HumanMessage(input));
      return { input };
    },
  }), # {input}
  # 注入摘要
  RunnablePassthrough.assign({
    maybe_summary_section: async () =>
      summary?.trim()
        ? `以下是之前对话摘要，请在回答时保持一致性：\n${summary}\n`
        : "",
  }), # {input, maybe_summary_section}
  chatPrompt, # 拼装提示词
  chatModel, # 调用模型
  new StringOutputParser(), # 解析输出
]);

async function chatLoop() {
  console.log("开始会话,输入内容后回车；输入/clear清空历史，/exit 退出");

  while (true) {
    # 设置输入输出的编码为 UTF-8
    readlineSync.setEncoding("utf8");
    const input = readlineSync.question("用户：").trim();

    if (!input) continue;

    if (input === "/exit") {
      console.log("拜拜");
      break;
    }

    if (input === "/clear") {
      summary = ""; # 清空摘要
      await histroy.clear();
      console.log("历史记录");
      continue;
    }

    try {
      let finalText = "";
      const stream = await chatChain.stream({ input });
      process.stdout.write("助理：");
      for await (const chunk of stream) {
        process.stdout.write(chunk);
        finalText += chunk;
      }
      console.log("\n");
      # 将模型范围的答案加入历史记录中
      await histroy.addMessage(new AIMessage(finalText));
      # 只获取本轮两条数据
      const lastTwo = (await histroy.getMessages()).slice(-2);
      # 通过本轮的消息对象
      const new_lines = lastTwo
        .map((m) => `${m.getType()}:${m.content}`)
        .join();
      summary = await summaryChain.invoke({
        summary,
        new_lines,
      });
      await histroy.clear();

      console.log("____当前摘要——————");
      console.log(summary?.trim() ? summary.trim() : "当前无摘要");
    } catch (err) {
      console.log(err);
    }
  }
}

chatLoop();
```

# 持久化记忆
* 1.存储至文件
* 2.存储至数据库

手动封装一个做持续化处理的FileChatMessageHistory类。
注意点：原子化写入、并发写入
```bash
import fs from "fs/promises";
import path from "path";
import { BaseListChatMessageHistory } from "@langchain/core/chat_history";
import { HumanMessage, AIMessage } from "@langchain/core/messages";

export class FileChatMessageHistory extends BaseListChatMessageHistory {
  constructor(filepath) {
    super();
    this.filepath = filepath; # 存储文件路径
    this._q = Promise.resolve(); # 串行队伍列
  }

  # 串行队列
  # fn 就是一个任务，——enqueue 就是该任务放置到队列里
  # 作用：放置同事读写文件导致数据不一致
  _enqueue(fn) {
    const run = () => fn();
    const p = this._q.then(run, run);
    this._q = p.catch(() => {}); # 保持队列不中断
    return p;
  }

  # 确保目录存在
  async _ensureDir() {
    await fs.mkdir(path.dirname(this.filepath), {
      recursive: true,
    });
  }

  # 原写入
  async _atomicWrite(text) {
    await this._ensureDir();
    const tmp =
      this.filepath +
      `.tmp-${Date.now()}-${Math.random().toString(36).slice(2)}`; # 临时文件目录地址
    await fs.writeFile(tmp, text, "utf8"); # 先写入到临时文件
    await fs.rename(tmp, this.filepath); # 进行原子替换
  }

  # 读取原始数据
  async _readRaw() {
    # [{type: 'human', content: 'xxx'}, {type: 'ai', content: 'xxxx'}]
    const txt = await fs.readFile(this.filepath, "utf-8");
    const arr = JSON.parse(txt);
    return arr;
  }

  # 保存消息
  async _saveMessages(messages) {
    const row = messages.map((m) => ({
      type: m.getType(),
      content: m.content,
    }));
    await this._atomicWrite(JSON.stringify(row, null, 2));
  }

  # 获取消息
  async getMessages() {
    const row = await this._readRaw();
    return row.map((r) =>
      r.type === "humam"
        ? new HumanMessage(r.content)
        : new AIMessage(r.content),
    );
  }

  # 添加一条消息
  async addMessage(message) {
    return this._enqueue(async () => {
      const msgs = await this.getMessages();
      msgs.push(message);
      await this._saveMessages(msgs);
    });
  }

  # 添加多条消息
  async addMessages(messages) {
    return this._enqueue(async () => {
      const msgs = await this.getMessages();
      msgs.push(...messages);
      console.log(messages);
      console.log(msgs);
      await this._saveMessages(msgs);
    });
  }

  # 情况
  async clear() {
    return this._enqueue(async () => {
      await this._atomicWrite("[]");
    });
  }
}
```
